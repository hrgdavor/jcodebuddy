The generator's own rules, in Java

A generated page is the last step of a pipeline, so the pipeline's rules are worth stating where a reader of the page can see them. Note the shape: a direct method call per case, so a stock IDE can follow the routing from entry point to leaf (DEC-019). 

// {@link hr.hrg.example.report.ReportPlanGenerator} builds the page plan a renderer consumes.
// {enabled:true, blockMarker: "implicit"}
package hr.hrg.example.report;

import java.util.List;

/** Turn a class index into an ordered page plan. No HTML here — presentation is the renderer's job. */
public final class ReportPlanGenerator {

    private final ClassIndex classIndex;

    public ReportPlanGenerator(ClassIndex classIndex) {
        this.classIndex = classIndex;
    }

    public PagePlan plan(String pageKind, List<String> typeNames) {
        List<LinkTarget> targets = typeNames.stream()
            .map(this::targetFor)
            .flatMap(Optional::stream)          // an unresolvable type is dropped, then reported
            .toList();
        return new PagePlan(pageKind, targets, rejected(typeNames, targets));
    }

    private Optional<LinkTarget> targetFor(String fqn) {
        return classIndex.find(fqn)                      // resolves by fully qualified name
            .map(entry -> new LinkTarget(fqn, entry.path(), entry.declarationLine()));
    }
}
